cmake_minimum_required(VERSION 3.24)
project(MidEngine VERSION 0.2.0 LANGUAGES CXX)
option(MID_ENGINE_BUILD_TESTS "Build Mid Engine native tests" ON)
add_library(mid_engine_core native/src/scene.cpp native/src/rendering.cpp native/src/pbr.cpp native/src/asset_manager.cpp native/src/editor_bridge.cpp native/src/physics.cpp native/src/animation.cpp native/src/script_runtime.cpp native/src/obj_importer.cpp native/src/mtl_importer.cpp)
add_library(MidEngine::Core ALIAS mid_engine_core)
target_compile_features(mid_engine_core PUBLIC cxx_std_20)
target_include_directories(mid_engine_core PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/native/include>)
set_target_properties(mid_engine_core PROPERTIES CXX_EXTENSIONS OFF POSITION_INDEPENDENT_CODE ON)
if(MSVC)
 target_compile_options(mid_engine_core PRIVATE /W4 /permissive- /EHsc)
else()
 target_compile_options(mid_engine_core PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
endif()
if(MID_ENGINE_BUILD_TESTS)
 enable_testing()
 foreach(test_name IN ITEMS scene rendering assets physics animation_script editor_bridge obj_importer pbr mtl_importer)
  if(test_name STREQUAL "scene")
   set(test_source native/tests/scene_tests.cpp)
  elseif(test_name STREQUAL "rendering")
   set(test_source native/tests/rendering_tests.cpp)
  elseif(test_name STREQUAL "assets")
   set(test_source native/tests/asset_manager_tests.cpp)
  elseif(test_name STREQUAL "physics")
   set(test_source native/tests/physics_tests.cpp)
  elseif(test_name STREQUAL "mtl_importer")
   set(test_source native/tests/mtl_importer_tests.cpp)
  elseif(test_name STREQUAL "pbr")
   set(test_source native/tests/pbr_tests.cpp)
  elseif(test_name STREQUAL "animation_script")
   set(test_source native/tests/animation_script_tests.cpp)
  elseif(test_name STREQUAL "editor_bridge")
   set(test_source native/tests/editor_bridge_tests.cpp)
  else()
   set(test_source native/tests/obj_importer_tests.cpp)
  endif()
  add_executable(mid_engine_${test_name}_tests ${test_source})
  target_link_libraries(mid_engine_${test_name}_tests PRIVATE MidEngine::Core)
  target_compile_features(mid_engine_${test_name}_tests PRIVATE cxx_std_20)
  add_test(NAME mid_engine.${test_name} COMMAND mid_engine_${test_name}_tests)
 endforeach()
endif()
