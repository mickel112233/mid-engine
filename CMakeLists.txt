cmake_minimum_required(VERSION 3.24)
project(MidEngine VERSION 0.2.0 LANGUAGES CXX)
option(MID_ENGINE_BUILD_TESTS "Build Mid Engine native tests" ON)
add_library(mid_engine_core native/src/scene.cpp native/src/rendering.cpp native/src/pbr.cpp native/src/asset_manager.cpp native/src/editor_bridge.cpp native/src/physics.cpp native/src/animation.cpp native/src/script_runtime.cpp native/src/obj_importer.cpp native/src/mtl_importer.cpp native/src/editor_session.cpp native/src/image_decoder.cpp native/src/model_importer.cpp native/src/lua_runtime.cpp native/src/opengl33_device.cpp native/src/native_editor_runtime.cpp)
add_library(MidEngine::Core ALIAS mid_engine_core)
find_package(assimp CONFIG QUIET)
if(TARGET assimp::assimp)
 target_link_libraries(mid_engine_core PRIVATE assimp::assimp)
 target_compile_definitions(mid_engine_core PRIVATE MID_ENGINE_WITH_ASSIMP=1)
elseif(TARGET assimp)
 target_link_libraries(mid_engine_core PRIVATE assimp)
 target_compile_definitions(mid_engine_core PRIVATE MID_ENGINE_WITH_ASSIMP=1)
endif()
find_package(PNG QUIET)
if(TARGET PNG::PNG)
 target_link_libraries(mid_engine_core PRIVATE PNG::PNG)
 target_compile_definitions(mid_engine_core PRIVATE MID_ENGINE_WITH_PNG=1)
endif()
find_package(JPEG QUIET)
if(TARGET JPEG::JPEG)
 target_link_libraries(mid_engine_core PRIVATE JPEG::JPEG)
 target_compile_definitions(mid_engine_core PRIVATE MID_ENGINE_WITH_JPEG=1)
endif()
find_package(SDL2 CONFIG QUIET)
if(TARGET SDL2::SDL2)
 target_sources(mid_engine_core PRIVATE native/src/native_desktop_host.cpp)
 target_link_libraries(mid_engine_core PRIVATE SDL2::SDL2)
 target_compile_definitions(mid_engine_core PUBLIC MID_ENGINE_WITH_SDL2=1)
elseif(TARGET SDL2::SDL2-static)
 target_sources(mid_engine_core PRIVATE native/src/native_desktop_host.cpp)
 target_link_libraries(mid_engine_core PRIVATE SDL2::SDL2-static)
 target_compile_definitions(mid_engine_core PUBLIC MID_ENGINE_WITH_SDL2=1)
endif()
if(TARGET SDL2::SDL2)
 set(MID_ENGINE_SDL2_TARGET SDL2::SDL2)
elseif(TARGET SDL2::SDL2-static)
 set(MID_ENGINE_SDL2_TARGET SDL2::SDL2-static)
endif()
if(MID_ENGINE_SDL2_TARGET)
 add_executable(mid_engine_desktop native/src/native_main.cpp)
 target_link_libraries(mid_engine_desktop PRIVATE MidEngine::Core ${MID_ENGINE_SDL2_TARGET})
 target_compile_features(mid_engine_desktop PRIVATE cxx_std_20)
endif()
find_package(Lua QUIET)
if(TARGET Lua::Lua)
 target_link_libraries(mid_engine_core PRIVATE Lua::Lua)
 target_compile_definitions(mid_engine_core PRIVATE MID_ENGINE_WITH_LUA=1)
elseif(LUA_FOUND)
 target_include_directories(mid_engine_core PRIVATE ${LUA_INCLUDE_DIR})
 target_link_libraries(mid_engine_core PRIVATE ${LUA_LIBRARIES})
 target_compile_definitions(mid_engine_core PRIVATE MID_ENGINE_WITH_LUA=1)
endif()
target_compile_features(mid_engine_core PUBLIC cxx_std_20)
target_include_directories(mid_engine_core PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/native/include>)
set_target_properties(mid_engine_core PROPERTIES CXX_EXTENSIONS OFF POSITION_INDEPENDENT_CODE ON)
if(MSVC)
 target_compile_options(mid_engine_core PRIVATE /W4 /permissive- /EHsc)
else()
 target_compile_options(mid_engine_core PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
endif()
if(MID_ENGINE_BUILD_TESTS)
 enable_testing()
 foreach(test_name IN ITEMS scene rendering assets physics animation_script editor_bridge editor_session image_decoder lua_runtime model_importer opengl33_device obj_importer pbr mtl_importer)
  if(test_name STREQUAL "scene")
   set(test_source native/tests/scene_tests.cpp)
  elseif(test_name STREQUAL "rendering")
   set(test_source native/tests/rendering_tests.cpp)
  elseif(test_name STREQUAL "assets")
   set(test_source native/tests/asset_manager_tests.cpp)
  elseif(test_name STREQUAL "physics")
   set(test_source native/tests/physics_tests.cpp)
  elseif(test_name STREQUAL "opengl33_device")
   set(test_source native/tests/opengl33_device_tests.cpp)
  elseif(test_name STREQUAL "model_importer")
   set(test_source native/tests/model_importer_tests.cpp)
  elseif(test_name STREQUAL "lua_runtime")
   set(test_source native/tests/lua_runtime_tests.cpp)
  elseif(test_name STREQUAL "image_decoder")
   set(test_source native/tests/image_decoder_tests.cpp)
  elseif(test_name STREQUAL "editor_session")
   set(test_source native/tests/editor_session_tests.cpp)
  elseif(test_name STREQUAL "mtl_importer")
   set(test_source native/tests/mtl_importer_tests.cpp)
  elseif(test_name STREQUAL "pbr")
   set(test_source native/tests/pbr_tests.cpp)
  elseif(test_name STREQUAL "animation_script")
   set(test_source native/tests/animation_script_tests.cpp)
  elseif(test_name STREQUAL "editor_bridge")
   set(test_source native/tests/editor_bridge_tests.cpp)
  else()
   set(test_source native/tests/obj_importer_tests.cpp)
  endif()
  add_executable(mid_engine_${test_name}_tests ${test_source})
  target_link_libraries(mid_engine_${test_name}_tests PRIVATE MidEngine::Core)
  target_compile_features(mid_engine_${test_name}_tests PRIVATE cxx_std_20)
  add_test(NAME mid_engine.${test_name} COMMAND mid_engine_${test_name}_tests)
  if(test_name STREQUAL "model_importer")
   set_tests_properties(mid_engine.model_importer PROPERTIES WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}")
  endif()
 endforeach()
endif()

# Installable SDK payload: core library plus public C++ headers.
include(GNUInstallDirs)
install(TARGETS mid_engine_core
  ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR}
  LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR}
  RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(DIRECTORY native/include/ DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
set(CPACK_PACKAGE_NAME "MidEngine")
set(CPACK_PACKAGE_VENDOR "Mid Engine Project")
set(CPACK_PACKAGE_VERSION ${PROJECT_VERSION})
set(CPACK_PACKAGE_DESCRIPTION_SUMMARY "Mid Engine native C++ engine core and public headers")
set(CPACK_PACKAGE_INSTALL_DIRECTORY "MidEngine")
include(CPack)
